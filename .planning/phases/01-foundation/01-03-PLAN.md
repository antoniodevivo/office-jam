    PART A: Docker Compose and Dockerfiles

    Step 1: Create docker-compose.yml at repo root with this EXACT content:
    ```yaml
    services:
      db:
        image: postgres:17-alpine
        environment:
          POSTGRES_DB: office_jam
          POSTGRES_USER: office_jam
          POSTGRES_PASSWORD: office_jam_dev
        ports:
          - "5432:5432"
        volumes:
          - pgdata:/var/lib/postgresql/data
        healthcheck:
          test: ["CMD", "pg_isready", "-U", "office_jam"]
          interval: 5s
          timeout: 5s
          retries: 5
          start_period: 10s

      api:
        build:
          context: .
          dockerfile: apps/api/Dockerfile
        depends_on:
          db:
            condition: service_healthy
        environment:
          DATABASE_URL: postgresql://office_jam:office_jam_dev@db:5432/office_jam
          HOST: 0.0.0.0
          PORT: "3000"
          NODE_ENV: production
        ports:
          - "3000:3000"

      web:
        build:
          context: .
          dockerfile: apps/web/Dockerfile
        depends_on:
          - api
        ports:
          - "4200:4200"

    volumes:
      pgdata:
    ```
    NOTE: No `version:` field -- it is obsolete in modern Docker Compose.

    Step 2: Create apps/api/Dockerfile. Read apps/api/project.json first to determine the build output directory (likely `dist/apps/api`). The Dockerfile:
    ```dockerfile
    FROM node:22-alpine AS builder
    WORKDIR /app
    COPY package.json package-lock.json ./
    RUN npm ci
    COPY . .
    RUN npx nx build api --configuration=production
    RUN cd libs/db && npx prisma generate

    FROM node:22-alpine AS runner
    WORKDIR /app
    COPY --from=builder /app/dist/apps/api ./dist/
    COPY --from=builder /app/libs/db/prisma ./prisma/
    COPY --from=builder /app/libs/db/prisma.config.ts ./prisma.config.ts
    COPY --from=builder /app/libs/db/generated ./generated/
    COPY --from=builder /app/node_modules ./node_modules/
    COPY --from=builder /app/package.json ./
    ENV NODE_ENV=production
    EXPOSE 3000
    CMD ["sh", "-c", "cd /app && npx prisma migrate deploy --config prisma/prisma.config.ts && node dist/main.js"]
    ```
    IMPORTANT: Adapt the COPY paths based on actual `nx build api` output. Read project.json to confirm the output directory. The CMD runs `prisma migrate deploy` first (applies existing migrations without creating new ones -- safe for production) then starts the server.

    Step 3: Create apps/web/Dockerfile. Read apps/web/project.json to determine build output:
    ```dockerfile
    FROM node:22-alpine AS builder

    COPY . .
    ```
