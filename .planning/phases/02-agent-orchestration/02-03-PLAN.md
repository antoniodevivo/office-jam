import type { BaseMessage } from "@langchain/core/messages";

// NOTE: LangGraph.js v1 uses Annotation for state definition.
// If StateSchema import is available, prefer it. If not, Annotation works.

export const WorkflowState = Annotation.Root({
  messages: Annotation<BaseMessage[]>({
    reducer: messagesStateReducer,

  taskId: Annotation<string>(),
  taskTitle: Annotation<string>(),

  iterationCount: Annotation<number>({

    default: () => 0,
  }),
  totalTokensUsed: Annotation<number>({
    reducer: (curr, update) => (update !== undefined ? update : curr),
    default: () => 0,

    reducer: (curr, update) => (update !== undefined ? update : curr),
    default: () => 10,
  }),
  maxTokenBudget: Annotation<number>({
    reducer: (curr, update) => (update !== undefined ? update : curr),
    default: () => 50000,
  }),
  startedAt: Annotation<number>({
    reducer: (curr, update) => (update !== undefined ? update : curr),
    default: () => Date.now(),

  maxExecutionMs: Annotation<number>({
    reducer: (curr, update) => (update !== undefined ? update : curr),
    default: () => 300000,
  }),
  status: Annotation<string>({
    reducer: (curr, update) => (update !== undefined ? update : curr),
    default: () => "running",

**Step 3: Create `apps/api/src/app/services/orchestration/guardrails.ts`:**

```typescript
import type { WorkflowStateType } from "./state";

export type GuardrailResult = "continue" | "timeout" | "max_iterations" | "budget_exceeded";

/**
 * Inspect workflow state and determine if any guardrail limit has been breached.
 *
 * This function is called in TWO places:
 * 1. By execute-task.ts AFTER graph.invoke() completes -- inspects the final
 *    state to detect if the workflow hit a limit during execution.
 * 2. Can also be used as a conditional edge inside custom StateGraph topologies
 *    (not used with createSupervisor since its topology is fixed).
 */
export function checkGuardrails(state: WorkflowStateType): GuardrailResult {
  const elapsed = Date.now() - state.startedAt;

  if (elapsed > state.maxExecutionMs) {
    return "timeout";
  }
  if (state.iterationCount >= state.maxIterations) {
    return "max_iterations";
  }
  if (state.totalTokensUsed >= state.maxTokenBudget) {
    return "budget_exceeded";
  }
  return "continue";
}
```

const ROLE_PROMPTS: Record<string, string> = {
  CEO: "You are the CEO. You provide strategic direction, make final decisions, and ensure the team stays aligned with the overall vision.",
  CTO: "You are the CTO. You make technical architecture decisions, evaluate technology choices, and guide the development strategy.",

  MARKETING: "You are a marketing specialist. You create marketing strategies, write copy, and plan campaigns.",
  LEGAL: "You are a legal advisor. You review compliance, draft policies, and identify legal risks.",
  CUSTOM: "You are a specialized team member. Follow your custom instructions carefully.",
};

export interface AgentNodeConfig {
  name: string;
  role: string;
  systemPrompt: string | null;
  model: BaseChatModel;
}

export function createAgentNode(config: AgentNodeConfig) {
  const { name, role, systemPrompt, model } = config;

  return createAgent({
    model,
    name,
    systemPrompt: fullPrompt,
    responseFormat: TaskOutputSchema,

}

Create `apps/api/src/app/services/orchestration/workflow.ts`:

This is the main orchestration entry point. It takes a task (with assignments including agents) and an office LLM config, then builds a compiled LangGraph workflow. It returns both the compiled graph AND the guardrail config so that the caller (execute-task.ts in Plan 04) can run post-invocation guardrail checks via `checkGuardrails()`.

**Why post-invocation guardrails instead of conditional edges:** `createSupervisor` from `@langchain/langgraph-supervisor` produces a fixed graph topology. You cannot inject custom conditional edges into its internal routing. Instead, guardrails are enforced by the caller: after `graph.invoke()` returns the final state, the caller calls `checkGuardrails(resultState)` to inspect `iterationCount`, `totalTokensUsed`, and elapsed time, then maps the result to the appropriate `TaskExecution.status` (e.g., `"timeout"`, `"budget_exceeded"`). LangGraph's built-in `recursionLimit` provides the hard safety net during invocation.

```typescript
import { createSupervisor } from "@langchain/langgraph-supervisor";
import { createAgentNode, AgentNodeConfig } from "./agent-node";
import { createChatModel, LlmFactoryParams } from "../llm-factory";
import { decrypt } from "../encryption";
import { DEFAULT_GUARDRAILS } from "@office-jam/shared";
import type { GuardrailConfig } from "@office-jam/shared";

// Types for the task and agent data from Prisma
export interface WorkflowAgent {
  id: string;
  name: string;
  role: string;
  systemPrompt: string | null;
  llmProvider: string;
  llmModel: string;
}

export interface WorkflowTask {
  id: string;
  title: string;
  description: string | null;
  assignments: Array<{ agent: WorkflowAgent }>;
}

export interface WorkflowOfficeLlmConfig {
  defaultProvider: string;
  defaultModel: string;
  openaiApiKey: string | null;
  anthropicApiKey: string | null;
  googleApiKey: string | null;
  ollamaBaseUrl: string;
}

/** Returned by buildWorkflow so the caller can enforce guardrails post-invocation */
export interface WorkflowResult {
  graph: any; // Compiled LangGraph graph -- typed as any because CompiledGraph generic varies
  guardrailConfig: GuardrailConfig;
}

function getApiKeyForProvider(
  provider: string,
  config: WorkflowOfficeLlmConfig
): string | null {
  switch (provider) {
    case "openai":
      return config.openaiApiKey ? decrypt(config.openaiApiKey) : null;
    case "anthropic":
      return config.anthropicApiKey ? decrypt(config.anthropicApiKey) : null;
    case "google":
      return config.googleApiKey ? decrypt(config.googleApiKey) : null;
    case "ollama":
      return null; // Ollama doesn't need API keys
    default:
      return null;
  }
}

export async function buildWorkflow(
  task: WorkflowTask,
  officeConfig: WorkflowOfficeLlmConfig,
  guardrails?: Partial<GuardrailConfig>
): Promise<WorkflowResult> {
  if (task.assignments.length === 0) {
    throw new Error("Task must have at least one agent assignment");
  }

  const guardrailConfig: GuardrailConfig = {
    ...DEFAULT_GUARDRAILS,
    ...guardrails,
  };

  // Create agent nodes from assignments
  const agentNodes = task.assignments.map(({ agent }) => {
    const apiKey = getApiKeyForProvider(agent.llmProvider, officeConfig);
    const model = createChatModel({
      provider: agent.llmProvider,
      model: agent.llmModel,
      apiKey,
      ollamaBaseUrl: officeConfig.ollamaBaseUrl,
    });

    return createAgentNode({
      name: agent.name.toLowerCase().replace(/\s+/g, "_"),
      role: agent.role,
      systemPrompt: agent.systemPrompt,
      model,
    });
  });

  // Create supervisor model (uses office default provider)
  const supervisorApiKey = getApiKeyForProvider(
    officeConfig.defaultProvider,
    officeConfig
  );
  const supervisorModel = createChatModel({
    provider: officeConfig.defaultProvider,
    model: officeConfig.defaultModel,
    apiKey: supervisorApiKey,
    ollamaBaseUrl: officeConfig.ollamaBaseUrl,
  });

  // Build supervisor workflow
  const workflow = createSupervisor({
    agents: agentNodes,
    llm: supervisorModel,
    prompt:
      `You are a project manager for the task: "${task.title}". ` +
      (task.description ? `Task details: ${task.description}. ` : "") +
      "Route work to the appropriate team member. " +
      "Synthesize their outputs into a final deliverable. " +
      "Delegate efficiently -- not every agent needs to be involved in every subtask.",
  });

  const graph = workflow.compile();

  // Return compiled graph + guardrail config.
  // The caller (execute-task.ts) will:
  // 1. Invoke graph with recursionLimit = guardrailConfig.maxIterations * 5
  // 2. After invoke returns, call checkGuardrails(resultState) to detect breaches
  // 3. Map the GuardrailResult to TaskExecution.status
  return { graph, guardrailConfig };
}
```

IMPORTANT IMPLEMENTATION NOTES:
1. If `createSupervisor` expects compiled agents (`.compile()` needed), add `.compile()` to each agent node. Test during implementation.
2. If `createAgent` / `createReactAgent` return type is incompatible with `createSupervisor` agents array, wrap them as raw StateGraph nodes instead. The research flagged this as open question #2.
3. The recursionLimit should be set when invoking the graph (in Plan 04), not during compilation.
4. **Guardrails are NOT conditional edges** -- `createSupervisor` topology is fixed. Guardrails are enforced post-invocation by the caller using `checkGuardrails()` from `guardrails.ts`. This is by design.
